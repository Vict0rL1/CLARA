/** Todo lo que apunta a un programado por id, y si resuelve hoy. */
function scheduleRefs(d: AppData): { ref: string; resolves: boolean }[] {
  const ids = new Set(d.schedules.map((s) => s.id))
  const out: { ref: string; resolves: boolean }[] = []
  for (const t of d.templates) if (t.kind === 'distribution') for (const l of t.lines) if (l.target.kind === 'payment') out.push({ ref: `template:${t.id}→${l.target.scheduleId}`, resolves: ids.has(l.target.scheduleId) })
  for (const sc of d.scenarios) for (const c of sc.changes) if (c.type === 'scheduleAmount') out.push({ ref: `scenario:${sc.id}→${c.scheduleId}`, resolves: ids.has(c.scheduleId) })
  for (const g of d.goals) if (g.plan?.link) out.push({ ref: `goal:${g.id}→${g.plan.link.scheduleId}`, resolves: ids.has(g.plan.link.scheduleId) })
  for (const s of d.inbox.snoozed) {
    const m = /^due:schedule:(.+):\d{4}-\d{2}-\d{2}$/.exec(s.id)
    if (m) out.push({ ref: `snooze:${s.id}`, resolves: ids.has(m[1]!) })
  }
  for (const s of d.inbox.dismissed) {
    const m = /^due:schedule:(.+):\d{4}-\d{2}-\d{2}$/.exec(s.id)
    if (m) out.push({ ref: `dismissed:${s.id}`, resolves: ids.has(m[1]!) })
  }
  for (const dist of d.incomeDistributions) for (const l of dist.lines) if (l.scheduleId) out.push({ ref: `distribution:${dist.id}→${l.scheduleId}`, resolves: ids.has(l.scheduleId) })
  for (const t of d.transactions) if (t.scheduleId) out.push({ ref: `tx:${t.id}→${t.scheduleId}`, resolves: ids.has(t.scheduleId) })
  return out
}
const dangling = (d: AppData) => scheduleRefs(d).filter((r) => !r.resolves).map((r) => r.ref)

describe('U · un rechazo nunca toca la entrada (congelada y serializada) en TODAS las ramas', () => {
  const check = (d: AppData, entryId: string, code: string, status?: string) => {
    const frozen = deepFreeze(persisted(d))
    const json = JSON.stringify(frozen)
    if (status) expect(canRevert(frozen, frozen.history.find((e) => e.id === entryId)!, ctx.today)).toBe(status)
    const r = revertEntry(frozen, entryId, ctx)
    expect(r.ok).toBe(false)
    expect(!r.ok && r.issues[0]!.code).toBe(code)
    expect(JSON.stringify(frozen)).toBe(json)
  }
  it('notFound', () => check(baseData(), 'nope', 'notFound'))
  it('notRevertible: corte replace, entrada sin cambios y entrada solo de aplazamientos', () => {
    check(baseData({ history: [{ id: 'r', at: NOW, source: 'replace', changes: [] }] }), 'r', 'notRevertible', 'notRevertible')
    check(baseData({ history: [{ id: 'e', at: NOW, source: 'app', changes: [] }] }), 'e', 'notRevertible', 'notRevertible')
    check(baseData({ history: [{ id: 's', at: NOW, source: 'app', changes: [{ collection: 'inboxSnoozed', id: 'due:x', before: null, after: { id: 'due:x', until: '2026-12-01', at: NOW } }] }] }), 's', 'notRevertible', 'notRevertible')
  })

  })
  it('conflict', () => {

  })
  it('currencyMismatch: un cambio de moneda hecho sin importes no se deshace cuando ya hay importes (y canRevert decía ok)', () => {

    const withMoney = recordHistory(changed, { ...changed, transactions: [tx({ id: 't1', currency: 'USD' })] }, NOW)
    check(withMoney, e.id, 'currencyMismatch', 'ok')
    // Sin importes sí se deshace y vuelve CAD.
    const r = revertEntry(changed, e.id, ctx)
    expect(r.ok && r.data.settings.currency).toBe('CAD')
  })

    const r1 = revertEntry(back, e1.id, ctx)
    expect(r1.ok && dangling(r1.data)).toEqual([])
    expect(r1.ok && r1.data.schedules.map((s) => s.id)).toEqual(['old'])

    const tracked = recordHistory(d, created.data, NOW)
    const e = tracked.history.at(-1)!
    // Posponer no crea entrada (R5); el aplazamiento caduca el 20 y hoy es 28: la bandeja lo ignora (`until > today`).
    const snoozed: AppData = { ...tracked, inbox: { snoozed: [{ id: 'due:schedule:s1:2026-09-01', until: '2026-09-20', at: EARLIER }], dismissed: [] } }
    expect(canRevert(snoozed, e, ctx.today)).toBe('ok')
    expect(revertDependents(snoozed, e, ctx.today)).toEqual([])
    expect(revertEntry(snoozed, e.id, ctx).ok).toBe(true)
  })
  it('un corte replace: no se revierte y «¿Qué cambió?» no reconstruye nada anterior', () => {
    const d = baseData({ history: [{ id: 'a', at: EARLIER, source: 'app', changes: [{ collection: 'accounts', id: 'main', before: null, after: account({ id: 'main' }) }] }, { id: 'r', at: NOW, source: 'replace', changes: [] }] })
    expect(canRevert(d, d.history[1]!, ctx.today)).toBe('notRevertible')
    expect(stateBefore(d, 0)).toBeNull()
  })
})

describe('S · reparación de particiones: huella del descarte, doble aplicación y rechazos sin cambios', () => {
  const TODAY = '2026-03-15'
  const c = at(TODAY)
  const audit = (): AppData =>
    baseData({
      accounts: [account({ id: 'main', name: 'Principal', anchor: { amountMinor: 180000, date: '2026-03-01', setAt: '2026-03-01T12:00:00.000Z' } })],
      schedules: [
        income('2026-01-31', 300000, { id: 'salary-old', name: 'Sueldo', frequency: 'monthly', endDate: '2026-02-27' }),
        income('2026-02-28', 320000, { id: 'salary-new', name: 'Sueldo', frequency: 'monthly' }),
      ],
    })
  const dismissed = () => {
    const r = dismissSplitRepair(audit(), 'salary-new', c)
    if (!r.ok) throw new Error('dismiss')
    return r.data
  }

  it('reparar dos veces: la segunda es notFound y no cambia nada; reparar un tramo que no cumple (serie sin partir) tampoco', () => {
    const d = audit()
    const r1 = repairSplit(d, 'salary-new', c)
    if (!r1.ok) throw new Error('r1')
    const frozen = deepFreeze(persisted(r1.data))
    const json = JSON.stringify(frozen)
    expect(repairSplit(frozen, 'salary-new', c)).toMatchObject({ ok: false, issues: [{ code: 'notFound' }] })
    expect(JSON.stringify(frozen)).toBe(json)
    expect(repairSplit(deepFreeze(persisted(d)), 'salary-old', c)).toMatchObject({ ok: false, issues: [{ code: 'notFound' }] })
    expect(repairSplit(deepFreeze(persisted(d)), 'nope', c)).toMatchObject({ ok: false, issues: [{ code: 'notFound' }] })
  })

  it('reparar solo añade anchorDate y updatedAt al tramo nuevo; el resto del registro y los demás registros son idénticos; queda una entrada revertible', () => {
    const d = audit()
    const r = repairSplit(d, 'salary-new', c)
    if (!r.ok) throw new Error('r')
    const before = d.schedules[1]!
    const { anchorDate, updatedAt, ...rest } = r.value
    expect([anchorDate, updatedAt]).toEqual(['2026-01-31', c.now])
    const { updatedAt: _u, ...restBefore } = before
    expect(rest).toEqual(restBefore)
    expect(r.data.schedules[0]).toBe(d.schedules[0])
    const tracked = recordHistory(d, r.data, c.now)
    expect(tracked.history.at(-1)!.changes.map((x) => [x.collection, x.id])).toEqual([['schedules', 'salary-new']])
    expect(canRevert(tracked, tracked.history.at(-1)!, TODAY)).toBe('ok')
  })

  it('descartar dos veces: la segunda es notFound (la tarjeta ya no lo propone)', () => {
    const d = dismissed()
    expect(dismissSplitRepair(deepFreeze(persisted(d)), 'salary-new', c)).toMatchObject({ ok: false, issues: [{ code: 'notFound' }] })
  })

  it('huella: cambiar la fecha del tramo de siempre (otro ancla raíz) vuelve a proponerlo; omitir fechas o cambiar el importe del tramo nuevo no (misma decisión)', () => {
    const d = dismissed()
    expect(d.inbox.dismissed[0]!.fingerprint).toBe(splitRepairFingerprint(d.schedules[1]!, '2026-01-31'))
    // El tramo de siempre pasa a fijarse el 30: la propuesta cambia (30-mar, 30-abr…) y la huella también.
    const rootChanged: AppData = { ...d, schedules: d.schedules.map((s) => (s.id === 'salary-old' ? { ...s, startDate: '2026-01-30' } : s)) }
    expect(driftedSplits(rootChanged, TODAY).map((x) => [x.scheduleId, x.anchor, x.proposed[0]])).toEqual([['salary-new', '2026-01-30', '2026-03-30']])
    // Omitir una fecha o cambiar el importe del tramo nuevo: sigue descartado (la decisión «cuenta desde el 28» no cambió).
    const skipped: AppData = { ...d, schedules: d.schedules.map((s) => (s.id === 'salary-new' ? { ...s, skippedDates: ['2026-03-28'], amountMinor: 1 } : s)) }
    expect(driftedSplits(skipped, TODAY)).toEqual([])
    // Partir el tramo nuevo otra vez con la versión corregida (28-abr): el tramo nuevo hereda anchorDate del 28, el descarte sigue valiendo para el tramo del 28.
    const r = saveScheduleFrom(d, 'salary-new', amountOnly(d.schedules[1]!, '2026-04-28', 330000, 'salary-3'), at('2026-04-20'))
    if (!r.ok) throw new Error(JSON.stringify(r.issues))
    expect(r.data.schedules.find((s) => s.id === 'salary-3')?.anchorDate).toBe('2026-02-28')
    expect(driftedSplits(r.data, '2026-04-20')).toEqual([])
  })

  it('el descarte cabe en el contrato de la bandeja con el id de programado más largo (64) y valida completo', () => {
    const longId = 'z'.repeat(64)
    const d = baseData({
      schedules: [
        income('2026-01-31', 300000, { id: 'salary-old', name: 'Sueldo', frequency: 'monthly', endDate: '2026-02-27' }),
        income('2026-02-28', 320000, { id: longId, name: 'Sueldo', frequency: 'monthly' }),
      ],
    })
    expect(driftedSplits(d, TODAY).map((x) => x.scheduleId)).toEqual([longId])
    const r = dismissSplitRepair(d, longId, c)
    if (!r.ok) throw new Error('dismiss')
    expect(r.data.inbox.dismissed[0]!.id).toBe(splitRepairDismissId(longId))
    expect(r.data.inbox.dismissed[0]!.id.length).toBeLessThanOrEqual(INBOX_ID_MAX)
    expect(validateAppData(persisted(r.data)).ok).toBe(true)
  })
})
