const at = (today: string) => ({ today, now: `${today}T12:00:00.000Z` })
function amountOnly(s: Schedule, startDate: string, amountMinor: number, id: string): ScheduleDraft {
  const { createdAt: _c, updatedAt: _u, currency: _cur, skippedDates: _k, ...rest } = s
  return { ...rest, id, startDate, amountMinor }
}
const split = (d: AppData, id: string, draft: ScheduleDraft, today: string) => {
  const r = saveScheduleFrom(d, id, draft, at(today))
  if (!r.ok) throw new Error(`split: ${JSON.stringify(r.issues)}`)
  return r
}
const figures = (d: AppData, today: string) => {
  const b = computeBudget(d, today)
  return [b.horizon?.endDate, b.reservedTotalMinor, b.availableMinor] as const
}
const main = (amountMinor: number, date: string) => account({ id: 'main', name: 'Principal', anchor: { amountMinor, date, setAt: `${date}T12:00:00.000Z` } })

/** Legacy (schema 13) amount-only split as 819eaa9 wrote it: old ends the day before, new starts on `from`, NO anchorDate. */
function legacySplit(old: Schedule, from: string, amountMinor: number, id: string, extra: Partial<Schedule> = {}): Schedule[] {
  const ended: Schedule = { ...old, endDate: addDaysStr(from, -1) }
  const next: Schedule = { ...old, id, startDate: from, amountMinor, endDate: undefined, skippedDates: [], ...extra }
  if (next.endDate === undefined) delete (next as Partial<Schedule>).endDate
  return [ended, next]
}
function addDaysStr(date: string, days: number): string {
  const [y, m, d] = date.split('-').map(Number) as [number, number, number]
  const t = new Date(Date.UTC(y, m - 1, d) + days * 86_400_000)
  return `${t.getUTCFullYear()}-${String(t.getUTCMonth() + 1).padStart(2, '0')}-${String(t.getUTCDate()).padStart(2, '0')}`
}

describe('E · F-03 exact percentage (analysis) and the other percent sites', () => {
  const MAX = MAX_AMOUNT_MINOR
  const seed = (previous: number[], current: number[]) =>
    baseData({
      accounts: [main(0, '2026-07-01')],
      transactions: [
        ...previous.map((amountMinor, i) => tx({ id: `p${i}`, date: '2026-08-10', amountMinor: Math.abs(amountMinor), kind: amountMinor < 0 ? 'refund' : 'expense', categoryId: 'groceries' })),
        ...current.map((amountMinor, i) => tx({ id: `c${i}`, date: '2026-09-10', amountMinor: Math.abs(amountMinor), kind: amountMinor < 0 ? 'refund' : 'expense', categoryId: 'groceries' })),
      ],
    })
  const change = (d: AppData) => analysisItems(d, '2026-09-28').find((i) => i.id === 'spendingUp' || i.id === 'spendingDown')

  it('E1 checkedMulDivFloor at the exact safe boundary: (90071992547409 × 100) is exact, one minor unit more is null', () => {
    expect(checkedMulDivFloor(90071992547409, 100, 1)).toBe(9007199254740900)
    expect(checkedMulDivFloor(90071992547410, 100, 1)).toBeNull()
    expect(checkedMulDivFloor(-90071992547409, 100, 1)).toBe(-9007199254740900)
    expect(checkedMulDivFloor(-90071992547410, 100, 1)).toBeNull()
    // floor semantics with negatives: −1/3 → −1; −3/3 → −1; 1/3 → 0.
    expect([checkedMulDivFloor(-1, 1, 3), checkedMulDivFloor(-3, 1, 3), checkedMulDivFloor(1, 1, 3)]).toEqual([-1, -1, 0])
  })

  it('E2 through analysisItems: previous 1 minor, current 900 719 925 474.10 → percent 9 007 199 254 740 900 exactly; one minor more → no figure', () => {
    const big = [...Array.from({ length: 90 }, () => MAX), 71992547410] // Σ = 90 071 992 547 410
    const up = change(seed([1], big))!
    expect(up.id).toBe('spendingUp')
    expect(up.params).toMatchObject({ percent: 9007199254740900, current: 90071992547410, previous: 1 })
    const over = change(seed([1], [...big, 1]))!
    expect(over.id).toBe('spendingUp')
    expect(over.params.percent).toBeUndefined()
    expect(over.params).toMatchObject({ current: 90071992547411, previous: 1 })
  })

  it('E3 large decrease: 100 × MAX → 0 is −100 % (spendingDown 100); 100 × MAX → 1 minor is also reported as 100 (floor of −99.99…)', () => {
    const prev = Array.from({ length: 100 }, () => MAX)
    expect(change(seed(prev, []))?.params).toMatchObject({ percent: 100, current: 0, previous: 100 * MAX })
    expect(change(seed(prev, []))?.id).toBe('spendingDown')
    expect(change(seed(prev, [1]))?.params).toMatchObject({ percent: 100, current: 1 })
  })

  it('E4 previous = 0 → no signal; previous negative (net refunds) → no signal; current negative with previous positive → spendingDown 150', () => {
    expect(change(seed([], [500000]))).toBeUndefined()
    expect(change(seed([2000, -5000], [500000]))).toBeUndefined()
    const down = change(seed([10000], [-5000]))!
    expect(down.id).toBe('spendingDown')
    expect(down.params).toMatchObject({ percent: 150, current: -5000, previous: 10000 })
    expect(down.because.find((b) => b.id === 'difference')?.value).toBe(-15000)
  })

  it('E5 threshold edges: +10.00 % fires, +9.99 % does not, −10.00 % fires, −9.99 % ALSO fires (floor rounds a decrease away from zero) and −9.00 % does not', () => {
    expect(change(seed([10000], [11000]))?.params).toMatchObject({ percent: 10 })
    expect(change(seed([10000], [10999]))).toBeUndefined()
    expect(change(seed([10000], [9000]))?.params).toMatchObject({ percent: 10 })
    // Documented formula floor(−999 × 100 / 10000) = floor(−9.99) = −10 → |−10| = 10 ≥ 10: the signal fires and says 10 %.
    expect(change(seed([10000], [9001]))?.params).toMatchObject({ percent: 10 })
    expect(change(seed([10000], [9100]))).toBeUndefined()
  })

  it('E6 plans.ts / categoryBreakdown use the unchecked mulDivFloor: a quotient above 2^53 comes back as an inexact Number (pre-existing, outside this diff)', () => {
    // spent 360 287 970 189 641 (odd) over a limit of 4 minor units: 9 007 199 254 741 025 (odd, > 2^53) cannot be a double.
    const q = mulDivFloor(360287970189641, 100, 4)
    expect(Number.isSafeInteger(q)).toBe(false)
    expect(checkedMulDivFloor(360287970189641, 100, 4)).toBeNull()
  })
})

describe('F · preservation (v14 splits and budget semantics)', () => {
  const data = (schedules: Schedule[], anchorDate = '2026-02-15', amountMinor = 100000) => baseData({ accounts: [main(amountMinor, anchorDate)], schedules })
  const salary = (startDate: string, extra: Partial<Schedule> = {}) => income(startDate, 300000, { id: 'salary', name: 'Sueldo', frequency: 'monthly', ...extra })

  it('F1 amount-only split with the bill ON the income day (31-mar) and on the would-be drifted day (28-mar): reservations never released, horizon stays 31-mar', () => {
    for (const billDate of ['2026-03-31', '2026-03-28']) {
      const d = data([salary('2026-01-31'), bill(billDate, 90000, { id: 'bill', name: 'Pago', frequency: 'once' })])
      expect(figures(d, '2026-03-20')).toEqual(['2026-03-31', 90000, 10000])
      const r = split(d, 'salary', amountOnly(d.schedules[0]!, '2026-02-28', 400000, 'next'), '2026-02-15')
      expect(figures(r.data, '2026-03-20')).toEqual(['2026-03-31', 90000, 10000])
      // Splitting exactly on the income day itself (from = 31-mar) keeps it too.
      const r2 = split(d, 'salary', amountOnly(d.schedules[0]!, '2026-03-31', 400000, 'next'), '2026-03-20')
      expect(figures(r2.data, '2026-03-20')).toEqual(['2026-03-31', 90000, 10000])
      expect(r2.value.next.anchorDate).toBe('2026-01-31')
    }
  })

  it('F2 repeated splits and re-saves keep the nominal 31 (monthly) and 31 (quarterly)', () => {
    const d = data([salary('2026-01-31')])
    const first = split(d, 'salary', amountOnly(d.schedules[0]!, '2026-02-28', 320000, 'n1'), '2026-02-15')
    const second = split(first.data, 'n1', amountOnly(first.value.next, '2026-04-30', 340000, 'n2'), '2026-04-10')
    expect(second.value.next).toMatchObject({ anchorDate: '2026-01-31', startDate: '2026-04-30' })
    expect(second.value.ended.endDate).toBe('2026-04-29')
    expect(occurrencesBetween(second.value.next, '2026-04-30', '2026-08-31')).toEqual(['2026-04-30', '2026-05-31', '2026-06-30', '2026-07-31', '2026-08-31'])
    // Plain re-save as the form does for scope «todas» (same startDate & rule → draft carries anchorDate): nothing moves.
    const draft: ScheduleDraft = { ...amountOnly(second.value.next, '2026-04-30', 345000, 'n2'), anchorDate: '2026-01-31', endDate: '2026-12-31' }
    const resaved = saveSchedule(second.data, draft, at('2026-05-01'))
    if (!resaved.ok) throw new Error(JSON.stringify(resaved.issues))
    expect(resaved.value).toMatchObject({ anchorDate: '2026-01-31', endDate: '2026-12-31' })
    expect(occurrencesBetween(resaved.value, '2026-05-01', '2026-12-31')).toEqual(['2026-05-31', '2026-06-30', '2026-07-31', '2026-08-31', '2026-09-30', '2026-10-31', '2026-11-30', '2026-12-31'])
    // Changing the startDate explicitly (the form drops anchorDate) re-anchors: an explicit change.
    const { anchorDate: _dropped, ...moveDraft } = amountOnly(second.value.next, '2026-05-05', 345000, 'n2')
    const moved = saveSchedule(second.data, moveDraft, at('2026-05-01'))
    if (!moved.ok) throw new Error(JSON.stringify(moved.issues))
    expect(moved.value.anchorDate).toBeUndefined()
    expect(occurrencesBetween(moved.value, '2026-05-01', '2026-07-31')).toEqual(['2026-05-05', '2026-06-05', '2026-07-05'])
    expect(driftedSplits(second.data, '2026-05-01')).toEqual([])
    expect(driftedSplits(resaved.data, '2026-05-01')).toEqual([])
    // Quarterly 31-oct: 2025-10-31 → 2026-01-31 → 04-30 → 07-31 → 10-31; split on 04-30 then on 10-31.
    const q = data([income('2025-10-31', 1000, { id: 'q', name: 'Bono', frequency: 'quarterly' })], '2025-01-01')
    const q1 = split(q, 'q', amountOnly(q.schedules[0]!, '2026-04-30', 1100, 'q1'), '2026-04-10')
    expect(occurrencesBetween(q1.value.next, '2026-04-30', '2027-01-31')).toEqual(['2026-04-30', '2026-07-31', '2026-10-31', '2027-01-31'])
    const q2 = split(q1.data, 'q1', amountOnly(q1.value.next, '2026-10-31', 1200, 'q2'), '2026-10-10')
    expect(q2.value.next.anchorDate).toBe('2025-10-31')
    expect(occurrencesBetween(q2.value.next, '2026-10-31', '2027-07-31')).toEqual(['2026-10-31', '2027-01-31', '2027-04-30', '2027-07-31'])
    expect(validateAppData(q2.data).ok).toBe(true)
  })

  it('F3 explicit changes re-anchor: rule change on a series date (monthly → quarterly from 31-mar) and an off-series date (30-mar); neither is later proposed for repair', () => {
    const d = data([salary('2026-01-31')])

    expect(driftedSplits(off.data, '2026-04-01')).toEqual([])
    // Semimonthly «1 y 20» → «1 y 15» from 20-oct (same frequency, other secondDay) re-anchors on 10-20: 20 and 5.
    const tw = data([bill('2026-08-01', 5000, { id: 'tw', name: 'Cuota', frequency: 'semimonthly', secondDay: 20 })])

  })
