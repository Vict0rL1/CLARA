const persisted = (d: AppData): AppData => JSON.parse(JSON.stringify(d)) as AppData
const at = (today: string) => ({ today, now: `${today}T12:00:00.000Z` })
function unwrap<T>(r: OpResult<T>): Extract<OpResult<T>, { ok: true }> {
  if (!r.ok) throw new Error(JSON.stringify(r.issues))
  return r
}
const imported = (text: string): AppData => {
  const r = parseBackup(text)
  if (!r.ok) throw new Error(JSON.stringify(r.issues))
  return r.data
}
const roundTrip = (d: AppData, today: string): AppData => imported(backupText(persisted(d), new Date(`${today}T12:00:00.000Z`), 'review'))
/** La misma copia marcada `schemaVersion: 13`: pasa por `migrate` (13 → 14 solo cambia el número). */
const migratedFrom13 = (d: AppData): AppData => {
  const r = validateAppData({ ...persisted(d), schemaVersion: 13 })
  if (!r.ok) throw new Error(JSON.stringify(r.issues))
  return r.data
}
const entryById = (d: AppData, id: string): HistoryEntry => {
  const e = d.history.find((x) => x.id === id)
  if (!e) throw new Error(`sin entrada ${id}`)
  return e
}
const lastEntry = (d: AppData): HistoryEntry => d.history[d.history.length - 1]!
const figures = (d: AppData, today: string) => {
  const b = computeBudget(d, today)
  return { available: b.availableMinor, reserved: b.reservedTotalMinor, overdue: b.overdueBills.map((i) => i.key) }
}
/** Edición directa de un programado, registrada en el historial como lo haría el formulario. */
const editSchedule = (d: AppData, id: string, patch: Partial<Schedule>, now: string): AppData =>
  recordHistory(d, { ...d, schedules: d.schedules.map((s) => (s.id === id ? { ...s, ...patch, updatedAt: now } : s)), updatedAt: now }, now)

/**
 * La negativa: las tres funciones coinciden, la lista es exactamente `expected`, y los datos (congelados en
 * profundidad) quedan byte a byte iguales.
 */
function expectRefused(data: AppData, entryId: string, today: string, expected: RevertDependent[]): void {
  const before = JSON.stringify(data)
  deepFreeze(data)
  const entry = entryById(data, entryId)
  expect(revertDependents(data, entry, today)).toEqual(expected)
  expect(canRevert(data, entry, today)).toBe('dependents')
  expect(revertEntry(data, entryId, at(today))).toEqual({ ok: false, issues: [{ path: 'history', code: 'revertDependents', params: { count: expected.length } }] })
  expect(JSON.stringify(data)).toBe(before)
}

describe('(a) entradas que solo tocan movimientos, y la papelera', () => {
  const relink = () => {
    const { data, repairEntry } = paid()
    const p1 = data.transactions.find((t) => t.id === 'p1')!
    const { createdAt: _c, updatedAt: _u, currency: _cur, realizedAt: _r, ...draft } = p1
    const r = unwrap(saveTransaction(data, { ...draft, occurrenceDate: '2026-04-30' }, at('2026-04-01')))
    const relinked = recordHistory(data, r.data, at('2026-04-01').now)
    return { relinked, relinkEntry: lastEntry(relinked), repairEntry }
  }

  it('revincular p1 del 31-mar al 30-abr y revertir: las dos fechas existen → se permite y el resultado valida', () => {
    const { relinked, relinkEntry } = relink()
    expect(relinkEntry.changes.map((c) => [c.collection, c.id])).toEqual([['transactions', 'p1']])
    expect(revertDependents(relinked, relinkEntry, '2026-04-01')).toEqual([])
    expect(canRevert(relinked, relinkEntry, '2026-04-01')).toBe('ok')
    const back = unwrap(revertEntry(relinked, relinkEntry.id, at('2026-04-01')))
    expect(back.data.transactions.find((t) => t.id === 'p1')?.occurrenceDate).toBe('2026-03-31')
    expect(validateAppData(persisted(back.data)).ok).toBe(true)
  })

  it('si después la serie pasa al día 30 (el 31-mar deja de existir): revertir la revinculación dejaría p1 sin fecha → se rechaza y se nombra la fecha ACTUAL del pago (30-abr)', () => {
    const { relinked, relinkEntry } = relink()
    const moved = editSchedule(relinked, 'b2', { anchorDate: '2026-01-30' }, '2026-04-02T12:00:00.000Z')
    expectRefused(moved, relinkEntry.id, '2026-04-02', [{ kind: 'payment', id: 'p1', name: '2026-04-30', scheduleId: 'b2', amountMinor: 42000 }])
  })

  it('eliminar el pago y revertir la eliminación: se permite; si entre medias se volvió a pagar la misma fecha, revertEntry lo devuelve pero validateAppData (la segunda guarda de la pantalla) rechaza la doble liquidación', () => {
    const { data } = paid()
    const del = unwrap(deleteTransaction(data, 'p1', at('2026-04-01')))
    const deleted = recordHistory(data, del.data, at('2026-04-01').now)
    const delEntry = lastEntry(deleted)
    expect(delEntry.changes.map((c) => [c.collection, c.id, c.after === null])).toEqual([['transactions', 'p1', true], ['trash', 'p1', false]])
    expect(revertDependents(deleted, delEntry, '2026-04-01')).toEqual([])
    expect(canRevert(deleted, delEntry, '2026-04-01')).toBe('ok')
    const back = unwrap(revertEntry(deleted, delEntry.id, at('2026-04-01')))
    expect(back.data.trash).toEqual([])
    expect(validateAppData(persisted(back.data)).ok).toBe(true)
    // Pagada de nuevo (p2) antes de revertir la eliminación de p1.
    const again = unwrap(markOccurrence(deleted, { scheduleId: 'b2', occurrenceDate: '2026-03-31', amountMinor: 42000, date: '2026-04-01', txId: 'p2' }, at('2026-04-01')))
    const repaid = recordHistory(deleted, again.data, at('2026-04-01').now)
    expect(revertDependents(repaid, delEntry, '2026-04-02')).toEqual([])
    expect(canRevert(repaid, delEntry, '2026-04-02')).toBe('ok')
    const twice = unwrap(revertEntry(repaid, delEntry.id, at('2026-04-02')))
    expect(twice.data.transactions.map((t) => t.id)).toEqual(['p-jan', 'p-feb', 'p2', 'p1'])
    const v = validateAppData(persisted(twice.data))
    expect(v.ok ? [] : v.issues).toEqual([{ path: 'transactions[3].occurrenceDate', code: 'duplicateId' }])
    // La operación de la papelera lo rechaza por sí misma.
    expect(restoreFromTrash(repaid, 'p1', at('2026-04-02'))).toMatchObject({ ok: false, issues: [{ code: 'occurrenceAlreadySettled' }] })
  })

  it('papelera: pagar, enviar el pago a la papelera, revertir la reparación (se permite) y restaurar el pago deja p1 apuntando a un 31-mar que ya no existe: 560 disponibles con el 28-mar vencido (el estado del P1, por otro camino)', () => {
    const { data, repairEntry } = paid()
    const del = unwrap(deleteTransaction(data, 'p1', at('2026-03-30')))
    const trashed = recordHistory(data, del.data, at('2026-03-30').now)
    expect(revertDependents(trashed, repairEntry, '2026-03-30')).toEqual([])
    expect(canRevert(trashed, repairEntry, '2026-03-30')).toBe('ok')
    const back = unwrap(revertEntry(trashed, repairEntry.id, at('2026-03-30')))
    expect(back.data.schedules.find((s) => s.id === 'b2')?.anchorDate).toBeUndefined()
    const restored = restoreFromTrash(back.data, 'p1', at('2026-03-30'))
    expect(restored.ok).toBe(true)
    if (!restored.ok) return
    expect(restored.data.transactions.find((t) => t.id === 'p1')).toMatchObject({ scheduleId: 'b2', occurrenceDate: '2026-03-31', status: 'realized' })
    expect(validateAppData(persisted(restored.data)).ok).toBe(true)
    expect(figures(restored.data, '2026-03-30')).toEqual({ available: 56000, reserved: 42000, overdue: ['schedule:b2:2026-03-28'] })
  })
})

describe('(c) tope de parejas de posibles duplicados (QA27-03b)', () => {
  const many = (ids: string[]) => ids.map((id) => tx({ id, date: '2026-09-10', amountMinor: 1200, kind: 'expense', status: 'realized' }))
  const ids = Array.from({ length: 60 }, (_, i) => `e${String(i).padStart(2, '0')}`)
  const FP = 'expense|main|1200|2026-09-10|~expense|main|1200|2026-09-10|'
  const dup = (x: AppData) => {
    const v = inboxView(x, TODAY)
    const only = (l: { kind: string; id: string }[]) => l.filter((i) => i.kind === 'duplicate').map((i) => i.id)
    return { active: only(v.active).length, snoozed: only(v.snoozed), dismissed: only(v.dismissed) }
  }
  const seeded = () => {
    const d = baseData({ transactions: many(ids) })
    const first = inboxView(d, TODAY).active.find((i) => i.id === 'dup:e00:e01')!
    const dismissed = unwrap(dismissInboxItem(d, first, ctx)).data
    return { d, first, both: unwrap(snoozeInboxItem(dismissed, 'dup:e00:e02', '2026-10-05', ctx)).data }
  }

  it('60 iguales el mismo día: 50 parejas estables (e00 con e01…e50) con la misma huella; descartar y posponer siguen aplicando con un movimiento más; cambiar la nota cambia la huella', () => {
    const { d, first, both } = seeded()
    const pairs = possibleDuplicates(d)
    expect(pairs).toHaveLength(DUPLICATE_PAIRS_PER_GROUP)
    expect(pairs.map(([a, b]) => `${a.id}:${b.id}`)).toEqual(ids.slice(1, 51).map((b) => `e00:${b}`))
    expect(possibleDuplicates(d).map(([a, b]) => `${a.id}:${b.id}`)).toEqual(pairs.map(([a, b]) => `${a.id}:${b.id}`))
    expect(first.fingerprint).toBe(FP)
    expect(validateInbox(both.inbox)).toEqual([])
    expect(dup(both)).toEqual({ active: 48, snoozed: ['dup:e00:e02'], dismissed: ['dup:e00:e01'] })
    const more = { ...both, transactions: [...both.transactions, ...many(['e60'])] }
    expect(dup(more)).toEqual({ active: 48, snoozed: ['dup:e00:e02'], dismissed: ['dup:e00:e01'] })
    const noted = { ...both, transactions: both.transactions.map((t) => (t.id === 'e01' ? { ...t, note: 'x' } : t)) }
    expect(dup(noted)).toEqual({ active: 49, snoozed: ['dup:e00:e02'], dismissed: [] })
  })

  it('un movimiento que ordena antes (a00) desplaza la ventana: las 50 parejas son otras y una pareja descartada o pospuesta deja de calcularse (ni activa ni descartada); su registro queda en `inbox` sin efecto', () => {
    const { both } = seeded()
    const shifted = { ...both, transactions: [...many(['a00']), ...both.transactions] }
    expect(dup(shifted)).toEqual({ active: 50, snoozed: [], dismissed: [] })
    expect(possibleDuplicates(shifted).every(([a]) => a.id === 'a00')).toBe(true)
    expect(shifted.inbox.dismissed.map((x) => x.id)).toEqual(['dup:e00:e01'])
    expect(shifted.inbox.snoozed.map((x) => x.id)).toEqual(['dup:e00:e02'])
  })
})
