const MAX = Number.MAX_SAFE_INTEGER // 9 007 199 254 740 991
const persisted = (d: AppData): AppData => JSON.parse(JSON.stringify(d)) as AppData
const at = (today: string) => ({ today, now: `${today}T12:00:00.000Z` })
const roundTrip = (d: AppData, today: string): AppData => {
  const r = parseBackup(backupText(persisted(d), new Date(`${today}T12:00:00.000Z`), 'test'))
  if (!r.ok) throw new Error(JSON.stringify(r.issues))
  return r.data
}
function must<T>(r: OpResult<T>): { data: AppData; value: T } {
  if (!r.ok) throw new Error(JSON.stringify(r.issues))
  return r
}
const figures = (d: AppData, today: string) => {
  const b = computeBudget(d, today)
  return {
    spendable: b.spendableMinor,
    horizon: b.horizon?.endDate ?? null,
    reserved: b.reservedTotalMinor,
    available: b.availableMinor,
    overdue: b.overdueBills.map((i) => i.key),
    overdueIncomes: b.overdueIncomes.map((i) => i.key),
  }
}
const lastEntry = (d: AppData): HistoryEntry => d.history.at(-1)!
const draftOf = (s: Schedule, patch: Partial<ScheduleDraft> = {}): ScheduleDraft => {
  const { createdAt: _c, updatedAt: _u, currency: _cur, skippedDates: _k, ...rest } = s
  return { ...rest, ...patch }
}
const track = (prev: AppData, next: AppData, today: string) => recordHistory(prev, next, at(today).now)
const pay = (d: AppData, input: Parameters<typeof markOccurrence>[1], today: string) => track(d, must(markOccurrence(d, input, at(today))).data, today)
const repaired = (d: AppData, id: string, today: string, nextId: string) => {
  const r = must(repairSplit(d, id, at(today), nextId))
  const data = track(d, r.data, today)
  return { data, entry: lastEntry(data), value: r.value }
}

/** `canRevert` y `revertEntry` dicen siempre lo mismo, y una negativa no cambia nada (igualdad JSON). */
const CODES = { notRevertible: 'notRevertible', alreadyReverted: 'alreadyReverted', conflict: 'revertConflict', dependents: 'revertDependents' } as const
function agree(data: AppData, entry: HistoryEntry, today: string) {
  const before = JSON.stringify(data)
  const status = canRevert(data, entry, today)
  const r = revertEntry(data, entry.id, at(today))
  expect(r.ok).toBe(status === 'ok')
  if (!r.ok) expect(r.issues[0]?.code).toBe(CODES[status as keyof typeof CODES])
  expect(JSON.stringify(data)).toBe(before)
  return status
}
type PaymentDependent = { id: string; scheduleId: string; name: string; amountMinor: number }
function expectBlockedBy(data: AppData, entry: HistoryEntry, today: string, payments: PaymentDependent[]) {
  expect(agree(data, entry, today)).toBe('dependents')
  expect(revertDependents(data, entry, today)).toEqual(payments.map((p) => ({ kind: 'payment', ...p })))
  expect(revertEntry(data, entry.id, at(today))).toMatchObject({ ok: false, issues: [{ path: 'history', code: 'revertDependents', params: { count: payments.length } }] })
}
/** Revertir permitido: el resultado valida entero (como al guardar) y se devuelve. */
function revertOk(data: AppData, entry: HistoryEntry, today: string): AppData {
  expect(agree(data, entry, today)).toBe('ok')
  const back = must(revertEntry(data, entry.id, at(today)))
  const v = validateAppData(persisted(back.data))
  expect(v.ok ? [] : v.issues).toEqual([])
  return back.data
}

/** Renta del 31 partida el 28-feb por una versión anterior (sin `anchorDate`); pagos del 28 en `paidDates`; saldo 10 000 el 1-ene; horizonte 30 días. */

    accounts: [account({ id: 'main', name: 'Principal', anchor: { amountMinor: 1000000, date: '2026-01-01', setAt: '2026-01-01T12:00:00.000Z' } })],
    schedules: [{ ...rent, endDate: '2026-02-27' }, { ...rent, id: 'r2', startDate: '2026-02-28', amountMinor: 160000, ...r2 }],
    transactions: [
      tx({ id: 'p0', kind: 'expense', amountMinor: 150000, date: '2026-01-31', categoryId: 'housing', scheduleId: 'r1', occurrenceDate: '2026-01-31' }),

describe('(a) QA27-01 · pagos sobre el tramo nuevo de una reparación PARTIDA', () => {
  it.each([
    ['adelantado', '2026-05-20', { reserved: 0, available: 210000 }],
    ['el mismo día', '2026-05-31', { reserved: 160000, available: 50000 }],
    ['tarde', '2026-06-03', { reserved: 160000, available: 50000 }],
  ])('pago %s del 31-may (registrado el %s): revertir la reparación se rechaza nombrando ese pago y su importe; nada cambia', (_label, date, expected) => {
    const { data, entry } = repaired(legacyRent(['2026-02-28', '2026-03-28', '2026-04-28']), 'r2', '2026-05-15', 'r3')
    expect(data.schedules.map((s) => [s.id, s.endDate ?? null, s.anchorDate ?? null])).toEqual([['r1', '2026-02-27', null], ['r2', '2026-04-30', null], ['r3', null, '2026-01-31']])
    const paid = pay(data, { scheduleId: 'r3', occurrenceDate: '2026-05-31', amountMinor: 160000, date, txId: 'p4' }, date)
    // 1 000 000 − 150 000 − 3 × 160 000 − 160 000 = 210 000; el 31-may pagado cuenta como movimiento, nunca como reserva.
    expect(figures(paid, date)).toMatchObject({ spendable: 210000, overdue: [], ...expected })
    expectBlockedBy(paid, entry, date, [{ id: 'p4', scheduleId: 'r3', name: '2026-05-31', amountMinor: 160000 }])
  })

  it('un pago tardío sobre el tramo que TERMINÓ (el 28-mar vencido, pagado el 16-may) no bloquea: la reversión lo conserva liquidando el 28-mar y el disponible es el mismo', () => {
    const d = legacyRent(['2026-02-28', '2026-04-28'])
    const TODAY = '2026-05-15'
    expect(driftedSplits(d, TODAY).map((f) => f.from)).toEqual(['2026-05-01'])
    // 1 000 000 − 150 000 − 2 × 160 000 = 530 000; reservados: 28-mar vencido + 28-may = 320 000.
    expect(figures(d, TODAY)).toEqual({ spendable: 530000, horizon: '2026-06-14', reserved: 320000, available: 210000, overdue: ['schedule:r2:2026-03-28'], overdueIncomes: [] })
    const { data, entry } = repaired(d, 'r2', TODAY, 'r3')
    expect(figures(data, TODAY)).toEqual({ spendable: 530000, horizon: '2026-06-14', reserved: 320000, available: 210000, overdue: ['schedule:r2:2026-03-28'], overdueIncomes: [] })
    const paid = pay(data, { scheduleId: 'r2', occurrenceDate: '2026-03-28', amountMinor: 160000, date: '2026-05-16', txId: 'p5' }, '2026-05-16')
    expect(figures(paid, '2026-05-16')).toEqual({ spendable: 370000, horizon: '2026-06-15', reserved: 160000, available: 210000, overdue: [], overdueIncomes: [] })
    const back = revertOk(paid, entry, '2026-05-16')
    expect(back.schedules.map((s) => [s.id, s.endDate ?? null])).toEqual([['r1', '2026-02-27'], ['r2', null]])
    expect(back.transactions.map((t) => t.id)).toEqual(['p0', 'p1', 'p2', 'p5'])
    expect(figures(back, '2026-05-16')).toEqual({ spendable: 370000, horizon: '2026-06-15', reserved: 160000, available: 210000, overdue: [], overdueIncomes: [] })
  })

  it('el tramo nuevo hereda la confirmación automática: el 31-may se registra solo y bloquea revertir la reparación; la entrada de la confirmación sí se revierte y el disponible no cambia (reserva o movimiento, nunca ambos)', () => {
    const { data, entry } = repaired(legacyRent(['2026-02-28', '2026-03-28', '2026-04-28'], { autoConfirm: true }), 'r2', '2026-05-15', 'r3')
    expect(data.schedules.find((s) => s.id === 'r3')?.autoConfirm).toBe(true)
    const c = must(autoConfirmDue(data, at('2026-05-31')))
    expect(c.value.confirmed.map((t) => [t.scheduleId, t.occurrenceDate, t.amountMinor, t.source])).toEqual([['r3', '2026-05-31', 160000, 'scheduled']])
    const id = c.value.confirmed[0]!.id
    const confirmed = track(data, c.data, '2026-05-31')
    expectBlockedBy(confirmed, entry, '2026-05-31', [{ id, scheduleId: 'r3', name: '2026-05-31', amountMinor: 160000 }])
    // Hoy 31-may, horizonte 30-jun: pagado → solo el 30-jun reservado; deshecho → 31-may y 30-jun reservados. Mismo disponible: 50 000.
    expect(figures(confirmed, '2026-05-31')).toEqual({ spendable: 210000, horizon: '2026-06-30', reserved: 160000, available: 50000, overdue: [], overdueIncomes: [] })
    const back = revertOk(confirmed, lastEntry(confirmed), '2026-05-31')
    expect(back.transactions.some((t) => t.id === id)).toBe(false)
    expect(figures(back, '2026-05-31')).toEqual({ spendable: 370000, horizon: '2026-06-30', reserved: 320000, available: 50000, overdue: [], overdueIncomes: [] })
  })
})

describe('(d) QA27-03b · tope de 50 parejas de posibles duplicados por grupo', () => {
  const pad = (i: number) => String(i).padStart(4, '0')
  const day = (prefix: string, date: string, n: number, extra: Partial<Transaction> = {}) =>
    Array.from({ length: n }, (_, i) => tx({ id: `${prefix}${pad(i)}`, date, amountMinor: 1200, kind: 'expense', status: 'realized', categoryId: 'groceries', ...extra }))
  const seed = (transactions: Transaction[]) => baseData({ accounts: money(0, '2026-08-01'), transactions })
  const ids = (d: AppData) => possibleDuplicates(d).map(([a, b]) => `${a.id}:${b.id}`)

  it('10 iguales el mismo día son 45 parejas (todas); 11 son 55 y se listan 50: todas las de t0000…t0006 más t0007:t0008', () => {
    expect(ids(seed(day('t', '2026-09-10', 10)))).toHaveLength(45)
    expect(ids(seed(day('t', '2026-09-10', 10)))).toContain('t0008:t0009')
    const eleven = ids(seed(day('t', '2026-09-10', 11)))
    expect(eleven).toHaveLength(DUPLICATE_PAIRS_PER_GROUP)
    expect(eleven.slice(0, 10)).toEqual(Array.from({ length: 10 }, (_, j) => `t0000:t${pad(j + 1)}`))
    expect(eleven.filter((p) => p.startsWith('t0006:'))).toEqual(['t0006:t0007', 't0006:t0008', 't0006:t0009', 't0006:t0010'])
    expect(eleven.at(-1)).toBe('t0007:t0008')
    for (const missing of ['t0007:t0009', 't0007:t0010', 't0008:t0009', 't0008:t0010', 't0009:t0010']) expect(eleven).not.toContain(missing)
  })

  it('las 50 son las mismas en cada cálculo y no cambian con un movimiento de otro importe; uno igual pero posterior entra como pareja de los primeros (orden fijo fecha, id) y uno anterior pasa delante', () => {
    const base = seed(day('t', '2026-09-10', 11))
    const first = ids(base)
    expect(ids(base)).toEqual(first)
    expect(ids(seed([...base.transactions, tx({ id: 'u', date: '2026-09-10', amountMinor: 1300, kind: 'expense', status: 'realized' })]))).toEqual(first)
    const later = ids(seed([...base.transactions, tx({ id: 'z', date: '2026-09-12', amountMinor: 1200, kind: 'expense', status: 'realized' })]))
    expect(later).toHaveLength(50)
    expect(later.slice(0, 10)).toEqual(first.slice(0, 10))
    expect(later[10]).toBe('t0000:z')
    const earlier = ids(seed([tx({ id: 'a', date: '2026-09-09', amountMinor: 1200, kind: 'expense', status: 'realized' }), ...base.transactions]))
    expect(earlier).toHaveLength(50)
    expect(earlier.slice(0, 11)).toEqual(Array.from({ length: 11 }, (_, j) => `a:t${pad(j)}`))
    expect(earlier.at(-1)).toBe('t0004:t0009')
    expect(earlier).not.toContain('t0007:t0008')
  })

  it('tres grupos (gasto, ingreso y otra cuenta): 50 + 3 + 1 parejas, cada una dentro de su tipo y cuenta', () => {
    const d = seed([...day('t', '2026-09-10', 11), ...day('i', '2026-09-10', 3, { kind: 'income', categoryId: 'salary' }), ...day('o', '2026-09-10', 2, { accountId: 'other' })])
    const pairs = possibleDuplicates(d)
    expect(pairs).toHaveLength(54)
    expect(pairs.filter(([a, b]) => a.kind === 'income' && b.kind === 'income')).toHaveLength(3)
    expect(pairs.filter(([a, b]) => a.accountId === 'other' && b.accountId === 'other')).toHaveLength(1)
    expect(pairs.filter(([a, b]) => a.kind === 'expense' && a.accountId === 'main' && b.kind === 'expense' && b.accountId === 'main')).toHaveLength(50)
  })

  it('ventana de ±3 días: a 3 días es pareja y a 4 no; con el tope, un segundo bloque del mismo importe diez días después no se lista nunca (observación)', () => {
    const edge = seed([tx({ id: 'a', date: '2026-09-10', amountMinor: 1200 }), tx({ id: 'b', date: '2026-09-13', amountMinor: 1200 }), tx({ id: 'c', date: '2026-09-14', amountMinor: 1200 })])
    expect(ids(edge)).toEqual(['a:b', 'b:c'])
    const pairs = possibleDuplicates(seed([...day('x', '2026-09-10', 60), ...day('y', '2026-09-20', 60)]))
    expect(pairs).toHaveLength(50)
    expect(pairs.every(([a, b]) => a.date === '2026-09-10' && b.date === '2026-09-10')).toBe(true)
  })

  it('pagos de la misma ocurrencia nunca son pareja ni consumen el tope: 6 parciales + 5 sueltos = 40 parejas, todas listadas; dos pagos del mismo programado en ocurrencias distintas tampoco (conservador)', () => {
    const d = seed([...day('l', '2026-09-10', 6, { scheduleId: 'rent', occurrenceDate: '2026-09-10', partialSettlement: true }), ...day('f', '2026-09-10', 5)])
    const pairs = possibleDuplicates(d)
    expect(pairs).toHaveLength(40)
    expect(pairs.some(([a, b]) => a.scheduleId === 'rent' && b.scheduleId === 'rent')).toBe(false)
    const other = seed([
      tx({ id: 'a', date: '2026-09-10', amountMinor: 1200, scheduleId: 'rent', occurrenceDate: '2026-08-31' }),
      tx({ id: 'b', date: '2026-09-11', amountMinor: 1200, scheduleId: 'rent', occurrenceDate: '2026-09-30' }),
    ])
    expect(ids(other)).toEqual([])
  })

  it('10 000 movimientos iguales el mismo día: 50 parejas, y `possibleDuplicates` + `inboxView` tardan menos de un segundo', () => {
    const d = seed(day('m', '2026-09-10', 10000))
    const t0 = performance.now()
    const pairs = possibleDuplicates(d)
    const view = inboxView(d, '2026-10-27')
    const elapsed = performance.now() - t0
    expect(pairs).toHaveLength(DUPLICATE_PAIRS_PER_GROUP)
    expect(view.active.filter((i) => i.kind === 'duplicate').map((i) => i.id).slice(0, 2)).toEqual(['dup:m0000:m0001', 'dup:m0000:m0002'])
    expect(elapsed).toBeLessThan(1000)
  })
})

describe('(e) · conservación: la copia v13 genuina, declinar, omitidas antes de `from`, colchón y rangos', () => {
  const V13 = readFileSync(new URL('../test/v13-known-intent-backup.json', import.meta.url), 'utf8')
  const imported = (): AppData => {
    const r = parseBackup(V13)
    if (!r.ok) throw new Error(JSON.stringify(r.issues))
    return r.data
  }
  const OVERDUE_INCOMES = ['schedule:salary:2026-01-31', 'schedule:salary-2:2026-02-28']

  it('copia v13 reparada el 21-mar (31-mar / 1 250 / 550): el 30-mar queda 1 día (550 por día y por semana); el sueldo del 31-mar recibido deja 2 957,77 y la reparación ya no se revierte, pero su cobro sí; igual tras recargar y en la copia', () => {
    const { data, entry } = repaired(imported(), 'salary-2', '2026-03-21', 'salary-3')
    expect(figures(data, '2026-03-21')).toEqual({ spendable: 180000, horizon: '2026-03-31', reserved: 125000, available: 55000, overdue: [], overdueIncomes: OVERDUE_INCOMES })
    const b30 = computeBudget(data, '2026-03-30')
    expect([b30.horizon?.endDate, b30.horizon?.days, b30.reservedTotalMinor, b30.availableMinor, b30.dailyMinor, b30.weeklyMinor, b30.weeklyDays]).toEqual(['2026-03-31', 1, 125000, 55000, 55000, 55000, 1])
    expect(b30.overdueBills.map((i) => i.key)).toEqual(['schedule:bill:2026-03-29'])
    const got = pay(data, { scheduleId: 'salary-2', occurrenceDate: '2026-03-31', amountMinor: 240777, date: '2026-03-31', txId: 'salary-mar' }, '2026-03-31')
    const after = { spendable: 420777, horizon: '2026-04-30', reserved: 125000, available: 295777, overdue: ['schedule:bill:2026-03-29'], overdueIncomes: OVERDUE_INCOMES }
    expect(figures(got, '2026-03-31')).toEqual(after)
    expectBlockedBy(got, entry, '2026-03-31', [{ id: 'salary-mar', scheduleId: 'salary-2', name: '2026-03-31', amountMinor: 240777 }])
    expect(agree(got, lastEntry(got), '2026-03-31')).toBe('ok')
    for (const again of [persisted(got), roundTrip(got, '2026-03-31')]) {
      expect(figures(again, '2026-03-31')).toEqual(after)
      expectBlockedBy(again, again.history.find((e) => e.id === entry.id)!, '2026-03-31', [{ id: 'salary-mar', scheduleId: 'salary-2', name: '2026-03-31', amountMinor: 240777 }])
    }
  })

  it('declinar («Dejar como está») sobrevive a exportar e importar: el 28 se conserva, nada se propone, las cifras no cambian y reparar ya no es posible', () => {
    const r = must(dismissSplitRepair(imported(), 'salary-2', at('2026-03-21')))

    expect(again.inbox.dismissed).toEqual([{ id: 'int:split:salary-2', fingerprint: '2026-02-28||2026-01-31', at: '2026-03-21T12:00:00.000Z' }])
